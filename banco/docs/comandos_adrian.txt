# Hilos del proceso: mismo PID, distinto LWP, estado y CPU
ps -L -o pid,lwp,comm,stat,%cpu -p $(pgrep -x carrera-0)

# Foto de top por hilo: mirar la linea %Cpu(s): "us" (Python) contra "sy" (kernel)
top -H -b -n 1 -p $(pgrep -x carrera-0) | head -20

# Cambios de contexto POR HILO (cada uno es una oportunidad para la carrera)
for T in /proc/$(pgrep -x carrera-0)/task/*; do echo "$(cat $T/comm): $(grep -h voluntary_ctxt $T/status | tr '\n' ' ')"; done

# Ojo: /proc/PID/status solo muestra los del hilo principal (por eso salen bajos)
grep ctxt /proc/$(pgrep -x carrera-0)/status

# Guardar la evidencia completa
sh observar.sh carrera_con_problema

###############################################################################
 B. DEMO DE ADRIAN MACIAS - ARQUITECTURA, PROCESOS E HILOS, MULTIPROCESO
###############################################################################

-------------------------------------------------------------------------------
 0. PREPARACION (antes de presentar)
-------------------------------------------------------------------------------

cd ~/Banco-operativos/banco
git pull origin main-olmpb1
rm -rf resultados evidencias

# Debe decir 2 o mas (ideal 4). Con 1 nucleo no se ve el paralelismo.
nproc

# Verificar que esta la version nueva (deben salir los dos archivos):
ls banco_con_problema.py banco_corregido.py

# Si el puerto 8000 quedo ocupado de antes:
fuser -k 8000/tcp


-------------------------------------------------------------------------------
 TERMINAL 1: arrancar el servidor (dejarla abierta)
-------------------------------------------------------------------------------

cd ~/Banco-operativos/banco
python3 servidor.py

# Navegador: http://localhost:8000
# Tarjeta 4 "Varios procesos trabajadores":
#   Procesos = 3    Hilos = 4    Operaciones por hilo = 50000


-------------------------------------------------------------------------------
 TERMINAL 2: todos los comandos de la demo
-------------------------------------------------------------------------------

cd ~/Banco-operativos/banco

# --- PASO 2: arquitectura -----------------------------------------------------

# Archivos del proyecto
ls *.py

# Las dos versiones tienen las mismas funciones
grep -n "^def " banco_con_problema.py banco_corregido.py

# Diferencia completa entre la version con problema y la corregida (salir con q)
diff banco_con_problema.py banco_corregido.py | less


# --- PASO 3: el proceso principal en reposo -------------------------------------

pstree -pt $(pgrep -f '^python3 servidor.py')


# --- PASOS 4 a 7: dar clic en "Ejecutar" (tarjeta 4) y ENSEGUIDA correr esto ---

# PASO 4: arbol de procesos e hilos (sin llaves = proceso, {llaves} = hilo)
pstree -pt $(pgrep -f '^python3 servidor.py')

# PASO 5: procesos hijo del servidor: PPID, numero de hilos, CPU y memoria
ps -o pid,ppid,nlwp,%cpu,rss,stat,etime,comm --ppid $(pgrep -f '^python3 servidor.py')

# PASO 6: los hilos de un proceso (mismo PID, distinto LWP)
ps -L -o pid,lwp,comm,stat,%cpu -p $(pgrep -x trabajador-0)

# PASO 7: memoria compartida: los 3 procesos mapean la MISMA region (rw-s)
for N in 0 1 2; do echo "trabajador-$N (PID $(pgrep -x trabajador-$N)):"; grep pym /proc/$(pgrep -x trabajador-$N)/maps; done

# Semaforos (locks) y memoria compartida en /dev/shm
ls -la /dev/shm/

# Guardar la evidencia completa en evidencias/
sh observar.sh multiproceso


-------------------------------------------------------------------------------
 PLAN B: todo junto de una sola vez
 Pegar este bloque completo apenas se da clic en "Ejecutar" (tarjeta 4)
-------------------------------------------------------------------------------

sleep 2; pstree -pt $(pgrep -f '^python3 servidor.py'); ps -o pid,ppid,nlwp,%cpu,rss,stat,etime,comm --ppid $(pgrep -f '^python3 servidor.py'); ps -L -o pid,lwp,comm,stat,%cpu -p $(pgrep -x trabajador-0); for N in 0 1 2; do echo "trabajador-$N (PID $(pgrep -x trabajador-$N)):"; grep pym /proc/$(pgrep -x trabajador-$N)/maps; done; sh observar.sh multiproceso

 PASO 4  banco-servidor con trabajador-0, trabajador-1, trabajador-2 como hijos;
